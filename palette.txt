# Solstice Crimson — palette reference
# Built on the original Solaris CDE "Crimson4" palette (palettes/Crimson4.dp
# in netrunner01/cde; also shipped in NsCDE's collection). A distinct member
# of the CDE Crimson family: deep crimson + sage chrome + warm sand canvas.
# Every core color below is byte-exact from the 48-bit original.

rose              #A93F6C   # slot 1, exact — selection, active titlebar, focus
chrome            #9CB9AF   # slot 2, exact — sage-grey panels, bars, menubars
                            # (gtk css + wallpaper generator base = desktop chrome)
steel             #65819C   # slot 3, exact — troughs, recessed, inactive border
canvas            #F2E9D9   # slot 4, exact — terminal / text-editor background
                            # (Omarchy terminal templates bind background here)
ink               #000000   # primary foreground
muted             #4A5953   # secondary text (Motif bottomShadow, darkened to 4.5:1)

# Motif-derived bevels from chrome (CDE computes these at runtime; no .dp source)
bevel_light       #D5E1DD   # raised highlight edges (topShadow)
shadow            #53635E   # deepest bevel shadows (bottomShadow)

# ANSI 16 (terminal)

color04  blue     #476899

color06  cyan     #466B6B
color07  white    #000000
